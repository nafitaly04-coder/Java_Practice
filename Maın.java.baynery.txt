
import java.util.Scanner;

public class Main {
    static void binary(int n) {
        if (n < 2) {
            System.out.print(n);
            return;
        }

        binary(n / 2);
        System.out.print(n % 2);
    }

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        int n = input.nextInt();

        if (n == 0) {
            System.out.println(0);
        } else {
            binary(n);
        }
    }
}
